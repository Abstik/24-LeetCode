~~~go
func isValid(s string) bool {
	n := len(s)
	if n == 0 {
		return true
	}
	stack := make([]int, 0, n)
	pp := make([]bool, n)
	for i := 0; i < n; i++ {
		if s[i] == '(' || s[i] == '[' || s[i] == '{' {
			stack = append(stack, i)
		} else if len(stack) > 0 {
			topIndex := len(stack) - 1
			topValue := stack[topIndex]
			if (s[i] == ')' && s[topValue] == '(') ||
				(s[i] == ']' && s[topValue] == '[') ||
				(s[i] == '}' && s[topValue] == '{') {
				pp[i] = true
				pp[topValue] = true
				stack = stack[:topIndex]
			}
		}
	}
	for i := 0; i < n; i++ {
		if !pp[i] {
			return false
		}
	}
	return true
}
~~~

~~~go
func removeDuplicates(s string) string {
	st := make([]byte, 0, len(s)) // append
	for i := 0; i < len(s); i++ {
		st = append(st, s[i])
		if len(st) == 1 {
			continue
		} else {
			if st[len(st)-2] == s[i] {
				st = st[:len(st)-2]
			}
		}
	}
	return string(st)
}
~~~

~~~go
func evalRPN(tokens []string) int {
	st := make([]string, 0, len(tokens))
	for _, token := range tokens {
		if token != "+" && token != "-" && token != "*" && token != "/" {
			st = append(st, token)
		} else {
			q, _ := strconv.Atoi(st[len(st)-1])
			p, _ := strconv.Atoi(st[len(st)-2])
			st = st[:len(st)-2]
			switch token {
			case "+":
				st = append(st, strconv.Itoa(p+q))
			case "-":
				st = append(st, strconv.Itoa(p-q))
			case "*":
				st = append(st, strconv.Itoa(p*q))
			case "/":
				st = append(st, strconv.Itoa(p/q))
			}
		}
	}
	rt, _ := strconv.Atoi(st[0])
	return rt
}
~~~

## [347. 前 K 个高频元素](https://leetcode.cn/problems/top-k-frequent-elements/)

~~~go
func topKFrequent(nums []int, k int) []int {
	mp := make(map[int]int)
	for _, n := range nums {
		mp[n]++
	}
	index := make([]struct {
		val int
		idx int
	}, 0, 100000)
	for i, v := range mp {
		index = append(index, struct {
			val int
			idx int
		}{v, i})
	}
	sort.Slice(index, func(i, j int) bool {
		return index[i].val > index[j].val
	})
	rt := []int{}
	for i := 0; i < k; i++ {
		rt = append(rt, index[i].idx)
	}
	return rt
}
~~~



## [239. 滑动窗口最大值](https://leetcode.cn/problems/sliding-window-maximum/)

![image-20250718095628374](C:\Users\18353\AppData\Roaming\Typora\typora-user-images\image-20250718095628374.png)



~~~go
type MyQueue struct {
	queue []int
}

func NewMyQueue() *MyQueue {
	return &MyQueue{
		queue: make([]int, 0),
	}
}

func (m *MyQueue) Front() int {
	return m.queue[0]
}

func (m *MyQueue) Back() int {
	return m.queue[len(m.queue)-1]
}

func (m *MyQueue) Empty() bool {
	return len(m.queue) == 0
}

func (m *MyQueue) Push(val int) {
	for !m.Empty() && val > m.Back() {
		m.queue = m.queue[:len(m.queue)-1]
	}
	m.queue = append(m.queue, val)
}

func (m *MyQueue) Pop(val int) {
	if !m.Empty() && val == m.Front() {
		m.queue = m.queue[1:]
	}
}

func maxSlidingWindow(nums []int, k int) []int {
	queue := NewMyQueue()
	length := len(nums)
	res := make([]int, 0)
	// 先将前k个元素放入队列
	for i := 0; i < k; i++ {
		queue.Push(nums[i])
	}
	// 记录前k个元素的最大值
	res = append(res, queue.Front())

	for i := k; i < length; i++ {
		// 滑动窗口移除最前面的元素
		queue.Pop(nums[i-k])
		// 滑动窗口添加最后面的元素
		queue.Push(nums[i])
		// 记录最大值
		res = append(res, queue.Front())
	}
	return res
}
~~~
