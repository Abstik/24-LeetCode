## [77. 组合 - 力扣（LeetCode）](https://leetcode.cn/problems/combinations/description/)

![image-20250716205619798](https://ting2.oss-cn-beijing.aliyuncs.com/picture/202507162056973.png)

```go
var ret [][]int   //定义返回数组
var arr []int     //定义每一组数

func combine(n int, k int) [][]int {
    arr = make([]int, 0, k)
    ret = make([][]int, 0)
    backtracking(n, k, 1)
    return ret
}

func backtracking(n int, k int, startIndex int){   //传入开始的下标
    //终止条件
    if len(arr) == k{    //个数够了，收集结果，返回
        t := make([]int, k)
        copy(t, arr)
        ret = append(ret, t)
        return 
    }
    for i:=startIndex; i<=n; i++{
        //为了保证数据不重复，找完1再找2只走34
        if n-i+1 < k-len(arr){   //剩下的元素不够k个了
            break   
        }
        arr = append(arr, i)
        backtracking(n, k, i+1)
        arr = arr[:len(arr) - 1]//弹出一个元素
    }
}
```
