## [98. 验证二叉搜索树 ](https://leetcode.cn/problems/validate-binary-search-tree/submissions/634230622/)

![image-20250603181418556](https://ting2.oss-cn-beijing.aliyuncs.com/picture/202506031814868.png)

```go
func isValidBST(root *TreeNode) bool {
    
    //思路一 把中序遍历的结果放入一个数组中，判断是不是升序
    //数组
    inorder := make([]int, 0, 10000)
    var dfs func(root *TreeNode)bool
    dfs = func(root *TreeNode)bool{
        if root == nil{
            return true  //如果是棵空树，是二叉搜索树
        }

        //中序遍历
        dfs(root.Left)  //先遍历左子树
        //中序遍历，将数据放入数组里
        inorder = append(inorder, root.Val)
        dfs(root.Right)  //遍历右子树

        return false
    }

    iforder := dfs(root)
    if iforder {
        return true
    }

    //判断数组是否有序，直接遍历
    for i:=1; i<len(inorder); i++{
        if inorder[i] <= inorder[i-1]{
            return false
        }
    }

    return true
    
    
    //思路二 直接在遍历的过程中判断元素是否单调递增
    //递归法  中序遍历
    //大于左子树所有元素，小于右子树所有元素
    var isValid func(root *TreeNode)bool
    maxVal := math.MinInt  //用来记录上一个结点的数值，来和当前比较

    isValid = func(root *TreeNode)bool{
        if root == nil {
            return true
        }
        left := isValid(root.Left)  //判断左子树是否符合要求
        if root.Val > maxVal{   //中
            maxVal = root.Val
        }else{
            return false   //当前结点不比前一个结点大，直接返回false
        }
        right := isValid(root.Right) //右子树

        //左右子树要同时符合条件才是二叉搜索树
        return left && right
    }
    ifValid := isValid(root)

    return ifValid
}
```

## [701. 二叉搜索树中的插入操作](https://leetcode.cn/problems/insert-into-a-binary-search-tree/)

![image-20250603183212850](https://ting2.oss-cn-beijing.aliyuncs.com/picture/202506031832976.png)

```go
func insertIntoBST(root *TreeNode, val int) *TreeNode {

    if root == nil{  //为空 直接赋值
        return &TreeNode{
            val,
            nil,
            nil,
        }
    }

    //查找插入位置
    if root.Val > val{  //左子树
        root.Left = insertIntoBST(root.Left, val)
    }else{
        root.Right =  insertIntoBST(root.Right, val)
    }

    return root
}
```
