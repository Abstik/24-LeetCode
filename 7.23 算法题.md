## [491. 非递减子序列 - 力扣（LeetCode）](https://leetcode.cn/problems/non-decreasing-subsequences/submissions/646443720/)

![image-20250723195754760](https://ting2.oss-cn-beijing.aliyuncs.com/picture/202507231957398.png)

```go
var(
    ret [][]int
    arr []int 
)

func findSubsequences(nums []int) [][]int {
    ret, arr = make([][]int, 0), make([]int, 0, len(nums))
    backtracking(nums, 0)
    return ret
}

func backtracking(nums []int, start int){

    if len(arr) >= 2{
        t := make([]int, len(arr))
        copy(t, arr)
        ret = append(ret, t)
    }

    used := make(map[int]bool, len(nums))   //用来记录有没有使用过
    for i:=start; i<len(nums); i++{   //数层去重
        if used[nums[i]] == true {    //去重，同一层使用过
            continue
        }
        if len(arr) == 0 || nums[i] >= arr[len(arr) - 1]{   //进入数组的条件，非递减
            arr = append(arr, nums[i])
            used[nums[i]] = true
            backtracking(nums, i+1)
            arr = arr[: len(arr) - 1]
        }
    }
}
```
