## [39. 组合总和 - 力扣（LeetCode）](https://leetcode.cn/problems/combination-sum/submissions/644542867/)

```go
var(
    ret [][]int   //定义返回数组
    arr []int     //每一个组合
)

func combinationSum(candidates []int, target int) [][]int {
    ret = make([][]int, 0)
    arr = make([]int, 0, len(candidates))
    sort.Ints(candidates)
    backtracking(candidates, target, 0)
    return ret
}

func backtracking(candidates []int, target int, startindex int){ //传入数组、目标和、开始下标
    if target == 0{  //终止条件  --  和为target
        t := make([]int, len(arr))
        copy(t, arr)
        ret = append(ret, t)
        return 
    }

    //继续回溯的逻辑
    for i:=startindex; i<len(candidates); i++{
        if candidates[i] > target{   //升序排序的
            break
        }
        arr = append(arr, candidates[i])  //把新的元素进数组
        backtracking(candidates, target - candidates[i], i)  //每次都从自己开始
        arr = arr[:len(arr) - 1]   //弹出元素
    }
}
```

```go
    if target == 0{  //终止条件  --  超出或者得到了想要的和

        copy(t, arr)
        ret = append(ret, t)
        return

        arr = arr[:len(arr) - 1]   //弹出
    }
```

## [131. 分割回文串 - 力扣（LeetCode）](https://leetcode.cn/problems/palindrome-partitioning/submissions/644670155/)

![image-20250717115554175](https://ting2.oss-cn-beijing.aliyuncs.com/picture/202507171155301.png)

```go
var(
    ret [][]string   //返回数组
    str []string    //每个数组
)

func partition(s string) [][]string {
    ret, str = make([][]string, 0, len(s)), make([]string, 0)
    backtracking(s, 0)
    return ret
}

//回溯
func backtracking(s string, startindex int){
    //终止条件 --  字符串中所有的字母都添加到了字符串里 -- 也就是起始位置到了最后
    if startindex == len(s){
        t := make([]string, len(str))
        copy(t, str)
        ret = append(ret, t)
        return 
    }

    for i:= startindex; i<len(s); i++{
        t := s[startindex:i+1]  //截取子串，从开始到现在 -- startindex -> i
        if ifPalindrome(t) {  //判断每一段是不是回文 -- 是回文（不是回文的话直接丢弃结果，不用再回溯了）
            str = append(str, t)  //一段是回文，保存进str
            backtracking(s, i+1)  //继续向后
            str = str[:len(str) -1]
        }
    }
}

//判断是否为回文串
func ifPalindrome(str string) bool {
    left := 0
    right := len(str)-1
    for left < right{
        if str[left] != str[right]{
            return false
        }
        left ++ 
        right -- 
    }
    return true
}
```

```go
var(
    ret []string 
    str []string   //先定义为字符串切片类型，最后通过.相连 
)

func restoreIpAddresses(s string) []string {
    
    ret = make([]string, 0)
    backtracking(s, 0)
    return ret
}

//回溯
func backtracking(s string, startindex int){   //传入数组和开始的下标

    //终止回溯的条件  --  已经有了四段
    if len(str) == 4{     //如果分成四段了，就一定不会向下遍历了   --   进一步判断是否复合要求即可
        if startindex == len(s){         //如果遍历到最后了，就是把原数组所有的都包含进去了，是符合要求
            sss := strings.Join(str, ".")     //用 . 连接
            ret = append(ret, sss)
        }
        return 
    }
    
    //继续回溯的处理逻辑
    for i:=startindex; i<len(s); i++{
        if i != startindex && s[startindex] == '0'{   //判断是否有前导0 
        //是从startindex截取到i+1
        //单独一个0  ---  i=startindex且s[startindex] = 0
        //前导 0   -----  第一个是0->s[startindex] = 0 ， 并且现在不是第一个数 ----> 有前导0 -- 直接退出
            break
        }
        ss := s[startindex : i+1]   //截取字符串
        num, _ := strconv.Atoi(ss)  //把截取的一段字符串转化为整型
        if num >= 0 && num <= 255{  //判断是否在0到255之间
            str = append(str, ss)
            backtracking(s, i+1)
            str = str[:len(str) - 1]
        }else{
            break
        }
    }
}
```
