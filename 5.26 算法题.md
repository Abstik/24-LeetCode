## [17. 电话号码的字母组合](https://leetcode.cn/problems/letter-combinations-of-a-phone-number/)

```go
var (
	index []string
	path  []byte
	rt    []string
)

func letterCombinations(digits string) []string {
	index = []string{"abc", "def", "ghi", "jkl", "mno", "pqrs", "tuv", "wxyz"}
	path, rt = make([]byte, 0), make([]string, 0)
	if digits == "" {
		return rt
	}
	dfs(digits, 0)
	return rt
}
func dfs(digits string, start int) {
	if len(path) == len(digits) { //终止条件，字符串长度等于digits的长度
		tmp := string(path)
		rt = append(rt, tmp)
		return
	}
	digit := int(digits[start] - '0') // 将index指向的数字转为int（确定下一个数字）
	str := index[digit-2]             // 取数字对应的字符集（注意和map中的对应）
	for j := 0; j < len(str); j++ {
		path = append(path, str[j])
		dfs(digits, start+1)
		path = path[:len(path)-1]
	}
}
```

## [39. 组合总和](https://leetcode.cn/problems/combination-sum/)

```go
var (
	rt   [][]int
	path []int
)

func combinationSum(candidates []int, target int) [][]int {
	rt, path = make([][]int, 0), make([]int, 0, len(candidates))
	sort.Ints(candidates) // 排序，为剪枝做准备
	dfs(candidates, 0, target)
	return rt
}

func dfs(candidates []int, start int, target int) {
	if target == 0 { // target 不断减小，如果为0说明达到了目标值
		tmp := make([]int, len(path))
		copy(tmp, path)
		rt = append(rt, tmp)
		return
	}
	for i := start; i < len(candidates); i++ {
		if candidates[i] > target { // 剪枝，提前返回
			break
		}
		path = append(path, candidates[i])
		dfs(candidates, i, target-candidates[i])
		path = path[:len(path)-1]
	}
}
```
