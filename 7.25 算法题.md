## [47. 全排列 II - 力扣（LeetCode）](https://leetcode.cn/problems/permutations-ii/description/)

![image-20250723203532896](https://ting2.oss-cn-beijing.aliyuncs.com/picture/202507232035229.png)

```go
var(
    ret [][]int 
    arr []int 
    used []int
)

func permuteUnique(nums []int) [][]int {
    sort.Ints(nums)
    ret, arr = make([][]int, 0), make([]int, 0, len(nums))
    used = make([]int, len(nums))
    backtracking(nums, 0)
    return ret
}

func backtracking(nums []int, start int){
    if len(arr) == len(nums){  //元素个数一样，收集结果
        t := make([]int, len(arr))
        copy(t, arr)
        ret = append(ret, t)
        return 
    } 

    for i:=0; i<len(nums); i++{
        if i!=0 && nums[i] == nums[i-1] && used[i-1] == 0{   //树层去重，和前一个元素一样
            continue
        }
        if used[i] == 0{   //树枝去重，没有被用过
            arr = append(arr, nums[i])
            used[i] = 1
            backtracking(nums, start+1)
            used[i] = 0
            arr = arr[:len(arr) - 1]
        }
    }
}
```
