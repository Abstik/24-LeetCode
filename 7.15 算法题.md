## [236. 二叉树的最近公共祖先 - 力扣（LeetCode）](https://leetcode.cn/problems/lowest-common-ancestor-of-a-binary-tree/submissions/643925733/)

![image-20250714211436704](https://ting2.oss-cn-beijing.aliyuncs.com/picture/202507142114023.png)

```go
//从底向顶找 -- 后序遍历
//该结点是否为空 -- 是否包含p或q

func lowestCommonAncestor(root, p, q *TreeNode) *TreeNode {
    if root == nil{
        return nil
    }
    if root == p || root == q{
        return root 
    }

    //后序遍历 -- 左右中
    //左子树中是否包含p或q
    left := lowestCommonAncestor(root.Left, p, q)
    //右子树中
    right := lowestCommonAncestor(root.Right, p, q)
    
    //如果左右子树都不为空 -- 该结点左子树和右子树包含了p和q -- 返回该节点
    if left != nil && right != nil{
        return root
    }else if left == nil && right != nil{  //左空 右不空 -- 右节点
        return right 
    }else if left != nil && right == nil{
        return left  
    }else {
        return nil
    }
}
```

## [235. 二叉搜索树的最近公共祖先 - 力扣（LeetCode）](https://leetcode.cn/problems/lowest-common-ancestor-of-a-binary-search-tree/submissions/643929655/)

![image-20250714212639575](https://ting2.oss-cn-beijing.aliyuncs.com/picture/202507142126689.png)

```go
//1.按照二叉树
//2.当前结点root与p和q结点的大小关系

func lowestCommonAncestor(root, p, q *TreeNode) *TreeNode {
    if root == nil{
        return nil
    }
    //1） root大于p和q->在root左子树
    if root.Val > p.Val && root.Val > q.Val{
        return lowestCommonAncestor(root.Left, p, q)
    }else if root.Val < p.Val && root.Val < q.Val{  //2） root小于p和q->在root右子树
        return lowestCommonAncestor(root.Right, p, q)
    }else{//3） root大于p小于q -> 是root 或者 root与p或q中有一个相等，也是root
        return root
    }
}
```

## [108. 将有序数组转换为二叉搜索树 - 力扣（LeetCode）](https://leetcode.cn/problems/convert-sorted-array-to-binary-search-tree/)

![image-20250715095214484](https://ting2.oss-cn-beijing.aliyuncs.com/picture/202507150952885.png)

```go
//中间的数作为根结点

func sortedArrayToBST(nums []int) *TreeNode {
    if len(nums) == 0{
        return nil
    }
    
    //找到中间结点的下标
    mid := len(nums)/2
    root := &TreeNode{
        Val : nums[mid],
    }

    //构造左子树 -- 左侧数组
    root.Left = sortedArrayToBST(nums[:mid])
    //构造右子树 -- 右侧数组
    root.Right = sortedArrayToBST(nums[mid+1:])

    return root
}
```

## [538. 把二叉搜索树转换为累加树 - 力扣（LeetCode）](https://leetcode.cn/problems/convert-bst-to-greater-tree/submissions/644018698/)

![image-20250715100637877](https://ting2.oss-cn-beijing.aliyuncs.com/picture/202507151006180.png)

```go
//中序遍历有序 -- 把自己及后面的所有值相加 -- 左中右
//从大往小遍历  --  右中左  --  上一个节点的值加上本节点的值

func convertBST(root *TreeNode) *TreeNode {

    prev := 0  //记录上一个节点的值
    //定义遍历函数
    var dfs func(node *TreeNode)
    dfs = func(node *TreeNode){
        if node == nil{
            return 
        }
        //遍历右子树
        dfs(node.Right)
        //处理当前节点
        node.Val += prev
        prev = node.Val  //更新prev节点的值
        //遍历左子树
        dfs(node.Left)
    }

    dfs(root)
    return root
}
```
