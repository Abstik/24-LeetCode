[Leetcode454. 四数相加 II ](https://leetcode.cn/problems/4sum-ii/)

``` go
func fourSumCount(nums1 []int, nums2 []int, nums3 []int, nums4 []int) int {
    sumMap := make(map[int]int)
    count := 0
    
    // 计算 nums1 和 nums2 所有可能的和及其出现次数
    for _, a := range nums1 {
        for _, b := range nums2 {
            sumMap[a+b]++
        }
    }
    
    // 计算 nums3 和 nums4 的所有可能的和，并检查其相反数是否存在于 sumMap 中
    for _, c := range nums3 {
        for _, d := range nums4 {
            target := -(c + d)
            count += sumMap[target] // 如果存在，则累加出现次数
        }
    }
    
    return count
}
```
