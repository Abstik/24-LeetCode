[Leetcode20.有效括号](https://leetcode.cn/problems/valid-parentheses/)

不匹配的有三种情况：1.左边括号无右边对应的 2.左右不匹配  3.右边多出来

```go
func isValid(s string) bool {
    stack :=make([]rune ,0)

    m:=make(map[rune]rune)
    m[')']='('
    m[']']='['
    m['}']='{'

    for _,c :=range s{
        if c=='('||c=='['||c=='{'{
            stack = append(stack,c)  //将左边括号放入栈中
        }else{
            if len(stack)==0{  //情况一：右边的括号有多余
                return false
            }
            peek :=stack[len(stack)-1]
            if peek !=m[c]{
                return false
            }
            stack =stack[:len(stack)-1]
        }
    }
    return len(stack)==0
}
```

```go
//写法二
func isValid(s string) bool{
    if len(s)%2!=0{
        return false
    }
    mp :=map[rune]rune{')':'(',']':'[','}':'{'}
    st :=[]rune{}
    for _,c:=range s{
        if mp[c]==0{  //c是左括号
            st.append(st,c)  //入栈
        }else{  //c是右括号
            if len(st)==0 ||st[len(st)-1]!=mp[c]{
                return false
            }
            st = st[:len(st)-1] //出栈
        }
    }
}
```
