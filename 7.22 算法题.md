## [78. 子集 - 力扣（LeetCode）](https://leetcode.cn/problems/subsets/submissions/645733918/)

![image-20250721163641086](https://ting2.oss-cn-beijing.aliyuncs.com/picture/202507211636563.png)

```go
var (
    ret [][]int  //返回数组
    arr []int    //每个数组
)


func subsets(nums []int) [][]int {
    ret, arr = make([][]int, 0), make([]int, 0, len(nums))
    backtracking(nums, 0)
    return ret
}

func backtracking(nums []int, start int){  //传入数组和开始的坐标
    //回溯终止条件 -- 到了数组最后
    t := make([]int, len(arr))
    copy(t, arr)
    ret = append(ret, t)

    for i:=start; i<len(nums); i++{
        arr = append(arr, nums[i])
        backtracking(nums, i+1)
        arr = arr[:len(arr) - 1]
    }
}
```

## [90. 子集 II - 力扣（LeetCode）](https://leetcode.cn/problems/subsets-ii/submissions/645928355/)

![image-20250722093400780](https://ting2.oss-cn-beijing.aliyuncs.com/picture/202507220934123.png)

```go
var(
    ret [][]int
    arr []int
)

func subsetsWithDup(nums []int) [][]int {
    ret, arr = make([][]int, 0), make([]int, 0)
    sort.Ints(nums)   //先排序
    backtracking(nums, 0)
    return ret
}

func backtracking(nums []int, start int){
    //每一个都要收集结果
    t := make([]int, len(arr))
    copy(t, arr)
    ret = append(ret, t)

    for i:=start; i<len(nums); i++{  //遍历
        if i != start && nums[i] == nums[i-1]{   //去重，树层去重 -- 如果不是第一个这个元素，跳过
            continue
        }
        arr = append(arr, nums[i])
        backtracking(nums, i+1)
        arr = arr[:len(arr) - 1]
    }
}
```
