~~~go
var (
	index []string
	path  []byte
	rt    []string
)

func letterCombinations(digits string) []string {
	index = []string{"abc", "def", "ghi", "jkl", "mno", "pqrs", "tuv", "wxyz"}
	path, rt = make([]byte, 0), make([]string, 0)
	if digits == "" {
		return rt
	}
	dfs(digits, 0)
	return rt
}
func dfs(idx string, start int) {
	if len(path) == len(idx) {
		tmp := string(path)
		rt = append(rt, tmp)
		return
	}
	digit := int(idx[start] - '0')
	str := index[digit-2]
	for j := 0; j < len(str); j++ {
		path = append(path, str[j])
		dfs(idx, start+1)
		path = path[:len(path)-1]
	}
}
~~~

## [39. 组合总和](https://leetcode.cn/problems/combination-sum/)

~~~go
var (
	rt   [][]int
	path []int
)

func combinationSum(candidates []int, target int) [][]int {
	rt, path = make([][]int, 0), make([]int, 0, len(candidates))
	sort.Ints(candidates)
	dfs(candidates, 0, target)
	return rt
}

func dfs(candidates []int, start int, target int) {
	if target == 0 {
		tmp := make([]int, len(path))
		copy(tmp, path)
		rt = append(rt, tmp)
		return
	}
	for i := start; i < len(candidates); i++ {
		if candidates[i] > target {
			break
		}
		path = append(path, candidates[i])
		dfs(candidates, i, target-candidates[i])
		path = path[:len(path)-1]
	}
}
~~~

~~~go
var (
	rt   [][]int
	path []int
)

func combinationSum2(candidates []int, target int) [][]int {
	rt, path = make([][]int, 0), make([]int, 0, len(candidates))
	sort.Ints(candidates)
	var dfs func([]int, int, int)
	dfs = func(candidates []int, start int, target int) {
		if target == 0 {
			tmp := make([]int, len(path))
			copy(tmp, path)
			rt = append(rt, tmp)
			return
		}
		for i := start; i < len(candidates); i++ {
			if candidates[i] > target {
				break
			}
			if i != start && candidates[i] == candidates[i-1] {
				continue
			}
			path = append(path, candidates[i])
			dfs(candidates, i+1, target-candidates[i])
			path = path[:len(path)-1]
		}
	}
	dfs(candidates, 0, target)
	return rt
}
~~~

## [131. 分割回文串](https://leetcode.cn/problems/palindrome-partitioning/)

![image-20250722092951562](C:\Users\18353\AppData\Roaming\Typora\typora-user-images\image-20250722092951562.png)



~~~go
var (
	path []string
	rt   [][]string
)

func partition(s string) [][]string {
	path, rt = make([]string, 0), make([][]string, 0)
	var dfs func(string, int)
	dfs = func(s string, start int) {
		if start == len(s) {
			tmp := make([]string, len(path))
			copy(tmp, path)
			rt = append(rt, tmp)
			return
		}
		for i := start; i < len(s); i++ {
			str := s[start : i+1]
			if judgeFunc(str) {
				path = append(path, str)
				dfs(s, i+1)
				path = path[:len(path)-1]
			}
		}
	}
	dfs(s, 0)
	return rt
}

func judgeFunc(s string) bool {
	for i, j := 0, len(s)-1; i < j; i, j = i+1, j-1 {
		if s[i] != s[j] {
			return false
		}
	}
	return true
}
~~~



## [93. 复原 IP 地址](https://leetcode.cn/problems/restore-ip-addresses/)

~~~go
var (
	path []string
	rt   []string
)

func restoreIpAddresses(s string) []string {
	path, rt = make([]string, 0, len(s)), make([]string, 0)
	var dfs func(string, int)
	dfs = func(s string, start int) {
		if len(path) == 4 {
			if start == len(s) {
				str := strings.Join(path, ".")
				rt = append(rt, str)
			}
			return
		}
		for i := start; i < len(s); i++ {
			if i != start && s[start] == '0' {
				break
			}
			str := s[start : i+1]
			num, _ := strconv.Atoi(str)
			if num >= 0 && num <= 255 {
				path = append(path, str)
				dfs(s, i+1)
				path = path[:len(path)-1]
			} else {
				break
			}
		}
	}
	dfs(s, 0)
	return rt
}
~~~
