## [860. 柠檬水找零](https://leetcode.cn/problems/lemonade-change/) 

![image-20250602214908177](https://gggzxh.oss-cn-beijing.aliyuncs.com/img/20250602214908177.png) 

```go
//定义两个数据表示5元和10元的剩余的数量，遍历数组，分别做判断
func lemonadeChange(bills []int) bool {
    five,ten:=0,0
    for _,v :=range bills {
        if v==5 {
            five++
        }else if v==10 {
            five--
            ten++
        }else if ten>0 {//如果是二十元且十元有剩余
            ten--
            five--
        }else {//没有十元则只减五元
            five-=3
        }
        if five<0 {//可以发现五元一定比十元减的快，如果五元没有了，则一定false
            return false
        }
    }
    return true
}
```
