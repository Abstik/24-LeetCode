

## [257. 二叉树的所有路径 - 力扣（LeetCode）](https://leetcode.cn/problems/binary-tree-paths/submissions/643831270/)

![image-20250714161304629](https://ting2.oss-cn-beijing.aliyuncs.com/picture/202507141613773.png)

```go
//回溯
func binaryTreePaths(root *TreeNode) []string {
    res := make([]string, 0)//定义返回数组
    var travel func(node *TreeNode, s string) //定义回溯函数travel
    travel = func(node *TreeNode, s string){ //s是一条路径
        if node.Left == nil && node.Right == nil{  //终止条件 -- 遍历到叶子结点
            v := s + strconv.Itoa(node.Val)  //Itoa将整数转换成字符串
            res = append(res, v)
            return 
        }
        s = s + strconv.Itoa(node.Val) + "->"  //构造题目给的形式
        //没有到叶子节点，继续遍历
        if node.Left != nil{
            travel(node.Left, s)
        }
        if node.Right != nil{
            travel(node.Right, s)
        }
    }

    //传入根结点 和 空的字符串
    travel(root, "")
    return res
}
```

## [617. 合并二叉树 - 力扣（LeetCode）](https://leetcode.cn/problems/merge-two-binary-trees/description/)

![image-20250714172201400](https://ting2.oss-cn-beijing.aliyuncs.com/picture/202507141722539.png)

```go
func mergeTrees(root1 *TreeNode, root2 *TreeNode) *TreeNode {
    //将第二棵二叉树合并到第一棵
    //都为空
    if root1 == nil && root2 == nil{
        return nil 
    }
    //左右都不空
    if root1 != nil && root2 != nil{
        root1.Val += root2.Val
    }
    //左空 右不空
    if root1 == nil && root2 != nil{
        root1 = &TreeNode{
            root2.Val,
            nil,
            nil,
        }
    }
    //左不空右空，不用处理
    if root1 != nil && root2 == nil{
        return root1
    }


    //处理左右子树
    root1.Left = mergeTrees(root1.Left, root2.Left)
    root1.Right = mergeTrees(root1.Right, root2.Right)

    return root1
}
```

## [530. 二叉搜索树的最小绝对差 - 力扣（LeetCode）](https://leetcode.cn/problems/minimum-absolute-difference-in-bst/submissions/643897582/)

![image-20250714194600508](https://ting2.oss-cn-beijing.aliyuncs.com/picture/202507141946857.png)

```go
//二叉搜索树的中序遍历--有序
//中序遍历的过程中记录差值

func getMinimumDifference(root *TreeNode) int {
    max := math.MaxInt
    prev := root.Val - max //记录上一个结点的值（确定初始值，既不能让root.Val作为最小值，也不能超出范围，使得最开始差值为最大即可）
    min := math.MaxInt  //记录差值

    //定义中序遍历的函数
    var inorder func(node *TreeNode)
    inorder = func(node *TreeNode){
        if node == nil{
            return 
        }
        //遍历左子树
        inorder(node.Left)
        //比较差值
        if node.Val - prev < min{
            min = node.Val - prev
        }
        prev = node.Val  //更新上一个结点
        //遍历右子树 
        inorder(node.Right)
    } 

    inorder(root)
    return min
}
```

## [501. 二叉搜索树中的众数 - 力扣（LeetCode）](https://leetcode.cn/problems/find-mode-in-binary-search-tree/submissions/643909709/)

![image-20250714202711628](https://ting2.oss-cn-beijing.aliyuncs.com/picture/202507142027025.png)

```go
//中序遍历，非递减

func findMode(root *TreeNode) []int {
    //定义返回数组
    var ret []int
    count := 1  //记录遍历过程中每个元素出现的次数
    maxCount := 1 //记录出现次数的最大值
    prev := 100001  //记录上一个元素的值
    //中序遍历
    var inorder func(root *TreeNode)
    inorder = func(root *TreeNode){
        if root == nil{
            return 
        }

        //遍历左子树
        inorder(root.Left)

        //处理当前节点
        if root.Val == prev{ //相等 ++ 
            count++   //当前元素等于上一个元素，count++
        }else{ //不等 count置为1
            count = 1
        }
        if count > maxCount && len(ret) > 0{   //更新最大值，同时判断ret中是否有元素，有元素清空
            ret = []int{root.Val}
            maxCount = count
        }else if count > maxCount && len(ret) == 0{
            ret = append(ret, root.Val)  //ret中没有元素，直接追加
            maxCount = count
        }else if count == maxCount{  //相等，直接加元素
            ret = append(ret, root.Val)
        }
        prev = root.Val  //更新prev

        //遍历右子树
        inorder(root.Right)
    }

    inorder(root)
    return ret
}
```
