~~~~go
func removeElement(nums []int, val int) int {
	i, j := 0, 0
	for ; j < len(nums); j++ {
		if nums[j] != val {
			nums[i] = nums[j]
			i++
		}
	}
	return i
}
~~~~

## [977. 有序数组的平方](https://leetcode.cn/problems/squares-of-a-sorted-array/)

~~~go
// 1.库函数法
func sortedSquares(nums []int) []int {
	for i := 0; i < len(nums); i++ {
		nums[i] = nums[i] * nums[i]
	}
	sort.Ints(nums)
	return nums
}

// 2.双指针法
func sortedSquares(nums []int) []int {
	i, j := 0, len(nums)-1
	k := len(nums) - 1
	rt := make([]int, len(nums))
	for i <= j {
		if nums[i]*nums[i] <= nums[j]*nums[j] {
			rt[k] = nums[j] * nums[j]
			k--
			j--
		} else {
			rt[k] = nums[i] * nums[i]
			k--
			i++
		}
	}
	return rt
}
~~~

## [209. 长度最小的子数组](https://leetcode.cn/problems/minimum-size-subarray-sum/)

![image-20250714202905591](C:\Users\18353\AppData\Roaming\Typora\typora-user-images\image-20250714202905591.png)



~~~go
func minSubArrayLen(target int, nums []int) int {
	cnt := 0
	rt := 100000
	for i, j := 0, 0; i < len(nums); i++ {
		cnt += nums[i]
		if cnt < target {
			continue
		}
		for cnt >= target {
			cnt -= nums[j]
			j++
			rt = int(math.Min(float64(rt), float64(i-j+1)))
		}
	}
	if rt == 100000 {
		return 0
	}
	return rt + 1
}
~~~



## [59. 螺旋矩阵 II](https://leetcode.cn/problems/spiral-matrix-ii/)

![image-20250714203848684](C:\Users\18353\AppData\Roaming\Typora\typora-user-images\image-20250714203848684.png)



~~~go
func generateMatrix(n int) [][]int {
	rt := make([][]int, n)
	for i := range rt {
		rt[i] = make([]int, n)
	}
	num := 1
	left, right, top, bottom := 0, n-1, 0, n-1
	for left <= right && top <= bottom {
		for col := left; col <= right; col++ {
			rt[top][col] = num
			num++
		}
		top++
		for row := top; row <= bottom; row++ {
			rt[row][right] = num
			num++
		}
		right--
		if top <= bottom {
			for col := right; col >= left; col-- {
				rt[bottom][col] = num
				num++
			}
			bottom--
		}
		if left <= right {
			for row := bottom; row >= top; row-- {
				rt[row][left] = num
				num++
			}
			left++
		}
	}
	return rt
}
~~~
