## [1005. K 次取反后最大化的数组和](https://leetcode.cn/problems/maximize-sum-of-array-after-k-negations/)

![image-20250528214202725](https://gggzxh.oss-cn-beijing.aliyuncs.com/img/20250528214202725.png)

```go
//在k次替换的同时，寻找每一次的最小值变化即可，遇到0就直接计算总和，可以避免修改大于0的数，使值最大
func findminIndex(nums []int) int {
    min:=nums[0]//寻找最小的数的下标，便于主函数改变
    index:=0
    for i:=1;i<len(nums);i++ {
        if nums[i]<min {
            min=nums[i]
            index=i
        }
    }
    return index 
}
func largestSumAfterKNegations(nums []int, k int) int {
    sum:=0
    for i:=0;i<k;i++{
        i:=findminIndex(nums)
        if nums[i]==0 {
            break
        }else {
            nums[i]*=-1
        }
    }
    for _,v :=range nums {
        sum+=v
    }
    return sum
}
```
