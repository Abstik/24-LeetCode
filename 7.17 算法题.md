~~~go
func removeElements(head *ListNode, val int) *ListNode {
	cur := head
	if head == nil {
		return head
	}
	for cur.Next != nil {
		if cur.Next.Val == val {
			cur.Next = cur.Next.Next
		} else {
			cur = cur.Next
		}
	}
	if head.Val == val {
		return head.Next
	}
	return head
}
~~~



## [206. 反转链表](https://leetcode.cn/problems/reverse-linked-list/)

~~~go
func reverseList(head *ListNode) *ListNode {
	var first, second, third *ListNode
	first, second, third = nil, head, nil
	for second != nil {
		third = second.Next
		second.Next = first
		first = second
		second = third
	}
	return first
}
~~~



## [142. 环形链表 II](https://leetcode.cn/problems/linked-list-cycle-ii/)

![image-20250717093959163](C:\Users\18353\AppData\Roaming\Typora\typora-user-images\image-20250717093959163.png)



~~~go
func detectCycle(head *ListNode) *ListNode {
    fast, slow := head, head
    pp := false
    
    for fast != nil && fast.Next != nil {
        fast = fast.Next.Next
        slow = slow.Next
        
        if slow == fast {
            pp = true
            break
        }
    }
    
    if pp {
        fast = head
        for fast != slow {
            fast = fast.Next
            slow = slow.Next
        }
        return slow
    }
    
    return nil
}
~~~



## [24. 两两交换链表中的节点](https://leetcode.cn/problems/swap-nodes-in-pairs/)![image-20250717094842389](C:\Users\18353\AppData\Roaming\Typora\typora-user-images\image-20250717094842389.png)



~~~go
func swapPairs(head *ListNode) *ListNode {
	if head == nil || head.Next == nil {
		return head
	}
	mid := head.Next
	head.Next = swapPairs(head.Next.Next)
	mid.Next = head
	return mid
}
~~~
