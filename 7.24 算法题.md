## [46. 全排列 - 力扣（LeetCode）](https://leetcode.cn/problems/permutations/submissions/646449736/)

![image-20250723201842544](https://ting2.oss-cn-beijing.aliyuncs.com/picture/202507232018896.png)

```go
var(
    ret [][]int 
    arr []int 
    used []int
)

func permute(nums []int) [][]int {
    ret, arr = make([][]int, 0), make([]int, 0, len(nums))
    used = make([]int, len(nums))
    backtracking(nums, 0)
    return ret 
}

func backtracking(nums []int, start int){
    if len(arr) == len(nums){    //元素个数和给定的数组元素个数相等，收集数据
        t := make([]int, len(nums))
        copy(t, arr)
        ret = append(ret, t)
        return 
    }

    for i:=0; i<len(nums); i++{
        if used[i] == 1 {    //已经使用过，跳过
            continue 
        }
        arr = append(arr, nums[i])  
        used[i] = 1   //设置已经用过
        backtracking(nums, start+1)
        used[i] = 0   //恢复原状太
        arr = arr[:len(arr) - 1]
    }
}
```
